cmake_minimum_required(VERSION 3.24)
project(NativeDispersion LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CUDA_STANDARD 17)
set(OPENMM_DIR "" CACHE PATH "Existing OpenMM environment prefix")
set(POCKETFFT_INCLUDE_DIR "" CACHE PATH "Directory containing pocketfft_hdronly.h")
find_package(CUDAToolkit REQUIRED)
add_library(NativeDispersion SHARED src/OpenMMForce.cpp src/CpuEngine.cpp src/CudaEngine.cu)
target_include_directories(NativeDispersion PRIVATE "${OPENMM_DIR}/include" "${OPENMM_DIR}/include/openmm/cuda" "${POCKETFFT_INCLUDE_DIR}")
target_link_libraries(NativeDispersion PRIVATE "${OPENMM_DIR}/lib/libOpenMM.so" "${OPENMM_DIR}/lib/plugins/libOpenMMCUDA.so" CUDA::cufft CUDA::cudart)
set_target_properties(NativeDispersion PROPERTIES BUILD_RPATH "${OPENMM_DIR}/lib;${OPENMM_DIR}/lib/plugins;${CUDAToolkit_LIBRARY_DIR}")
target_compile_options(NativeDispersion PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-O3> $<$<COMPILE_LANGUAGE:CUDA>:-O3>)
enable_testing()
add_executable(TestNativeDispersion tests/TestNativeDispersion.cpp)
target_include_directories(TestNativeDispersion PRIVATE src)
target_link_libraries(TestNativeDispersion PRIVATE NativeDispersion "${OPENMM_DIR}/lib/libOpenMM.so" "${OPENMM_DIR}/lib/plugins/libOpenMMCUDA.so" CUDA::cuda_driver)
add_test(NAME TestNativeDispersion COMMAND TestNativeDispersion)
